import * as THREE from "three";

/* ---------- ESCENA ---------- */
const canvas = document.getElementById("c");
const renderer = new THREE.WebGLRenderer({canvas, antialias:true, alpha:true});
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(38, 1, 0.1, 100);
camera.position.set(0, 0, 8);
scene.add(new THREE.AmbientLight(0xffffff, 1.0));
const key = new THREE.DirectionalLight(0xffffff, 2.4); key.position.set(3, 5, 5); scene.add(key);
const fill = new THREE.DirectionalLight(0x88aaff, 0.9); fill.position.set(-5, -2, 3); scene.add(fill);

const COLORS = {px:0xd62828, nx:0xff7a00, py:0xf4f4f4, ny:0xffd60a, pz:0x2ca02c, nz:0x1f6fe5};
const bodyGeo = new THREE.BoxGeometry(0.96, 0.96, 0.96);
const bodyMat = new THREE.MeshStandardMaterial({color:0x0b0b10, roughness:0.55});
const stGeo = new THREE.PlaneGeometry(0.80, 0.80);
const stMats = {};
for (const k in COLORS) stMats[k] = new THREE.MeshStandardMaterial({color:COLORS[k], roughness:0.35, metalness:0.05});

function makeCubie(x, y, z){
  const g = new THREE.Group();
  const bm = bodyMat.clone(); g.userData.mats = [bm];
  g.add(new THREE.Mesh(bodyGeo, bm));
  const sx = Math.sign(x), sy = Math.sign(y), sz = Math.sign(z), o = 0.481;
  const faces = [
    [sx>0?"px":"nx", [sx*o,0,0], [0, sx*Math.PI/2, 0]],
    [sy>0?"py":"ny", [0,sy*o,0], [-sy*Math.PI/2, 0, 0]],
    [sz>0?"pz":"nz", [0,0,sz*o], [0, sz>0?0:Math.PI, 0]],
  ];
  for (const [k, p, r] of faces){
    const sm = stMats[k].clone(); g.userData.mats.push(sm);
    const m = new THREE.Mesh(stGeo, sm);
    m.position.set(...p); m.rotation.set(...r); g.add(m);
  }
  g.position.set(x, y, z);
  return g;
}

const floater = new THREE.Group();   // sube y baja
const cubeGroup = new THREE.Group(); // inclinación del cubo
floater.add(cubeGroup); scene.add(floater);
const cubies = [];
for (const x of [-0.5,0.5]) for (const y of [-0.5,0.5]) for (const z of [-0.5,0.5]){
  const c = makeCubie(x,y,z); cubies.push(c); cubeGroup.add(c);
}

// sombra suave en el "suelo" para dar sensación de flotar
const sc = document.createElement("canvas"); sc.width = sc.height = 128;
const sctx = sc.getContext("2d");
const grad = sctx.createRadialGradient(64,64,0,64,64,64);
grad.addColorStop(0,"rgba(0,0,0,.55)"); grad.addColorStop(1,"rgba(0,0,0,0)");
sctx.fillStyle = grad; sctx.fillRect(0,0,128,128);
const shadow = new THREE.Mesh(new THREE.PlaneGeometry(3.2,3.2),
  new THREE.MeshBasicMaterial({map:new THREE.CanvasTexture(sc), transparent:true, depthWrite:false}));
shadow.rotation.x = -Math.PI/2; shadow.position.set(0,-2.1,0); scene.add(shadow);

/* ---------- EJES X, Y, Z (giran con el cubo) ---------- */
const axesG = new THREE.Group(); cubeGroup.add(axesG);
const AXCOL = {x:0xff4d4d, y:0x4dff88, z:0x4da3ff};
function label(txt, col){
  const c = document.createElement("canvas"); c.width = c.height = 128; const g = c.getContext("2d");
  g.font = "bold 92px system-ui"; g.textAlign = "center"; g.textBaseline = "middle";
  g.fillStyle = "#" + col.toString(16).padStart(6,"0"); g.fillText(txt, 64, 70);
  const sp = new THREE.Sprite(new THREE.SpriteMaterial({map:new THREE.CanvasTexture(c), depthTest:false, transparent:true}));
  sp.scale.setScalar(0.55); sp.renderOrder = 10; return sp;
}
for (const [k, v] of [["x",[1,0,0]],["y",[0,1,0]],["z",[0,0,1]]]){
  const dir = new THREE.Vector3(...v), col = AXCOL[k];
  const line = new THREE.Mesh(new THREE.CylinderGeometry(0.028,0.028,3.3,8), new THREE.MeshBasicMaterial({color:col}));
  line.quaternion.setFromUnitVectors(new THREE.Vector3(0,1,0), dir);
  const cone = new THREE.Mesh(new THREE.ConeGeometry(0.1,0.28,12), new THREE.MeshBasicMaterial({color:col}));
  cone.position.copy(dir).multiplyScalar(1.78); cone.quaternion.copy(line.quaternion);
  const lb = label(k.toUpperCase(), col); lb.position.copy(dir).multiplyScalar(2.15);
  axesG.add(line, cone, lb);
}

/* ---------- AGARRE CON PELLIZCO: capas y órbita ---------- */
const $ = id => document.getElementById(id);
const ptrEl = $("ptr"), video = $("video");
const target = {x:0,y:0}, cur = {x:0,y:0};
let ptrVisible = false, lastHand = 0, lastMouse = 0, handSeenOnce = false;
let pinched = false, wasPinched = false;
let yaw = -0.6, pitch = 0.5;           // orientación del cubo (quieto salvo que lo muevas)
let grab = null, snap = null, glowKey = "";
const HALF = Math.PI/2;
const ease = t => t<.5 ? 4*t*t*t : 1-Math.pow(-2*t+2,3)/2;
const ray = new THREE.Raycaster(), v2 = new THREE.Vector2();

const AX = [new THREE.Vector3(1,0,0), new THREE.Vector3(0,1,0), new THREE.Vector3(0,0,1)];
function pick(x, y){
  v2.set(x/innerWidth*2-1, -(y/innerHeight)*2+1);
  ray.setFromCamera(v2, camera);
  const hits = ray.intersectObjects(cubies, true);
  if (!hits.length) return null;
  const hit = hits[0];
  let o = hit.object; while (o && !cubies.includes(o)) o = o.parent;
  if (!o) return null;
  // eje local del cubo al que es perpendicular la cara tocada
  const nl = hit.face.normal.clone().transformDirection(hit.object.matrixWorld)
               .applyQuaternion(cubeGroup.quaternion.clone().invert());
  const n = [Math.abs(nl.x), Math.abs(nl.y), Math.abs(nl.z)];
  return {cubie:o, point:hit.point, normalAxis:n.indexOf(Math.max(...n))};
}
const coord = (c, i) => i===0 ? c.position.x : i===1 ? c.position.y : c.position.z;
function setGlow(list){
  const key = list ? list.map(c=>cubies.indexOf(c)).join() : "";
  if (key === glowKey) return; glowKey = key;
  cubies.forEach(c => { const on = !!list && list.includes(c);
    c.userData.mats.forEach(m => m.emissive.setHex(on ? 0x1d6f8c : 0x000000)); });
}
function makePivot(axisIdx, sideSign){
  const layer = cubies.filter(c => coord(c, axisIdx)*sideSign > 0);
  const pivot = new THREE.Group(); cubeGroup.add(pivot);
  layer.forEach(c => pivot.attach(c));
  return {layer, pivot};
}
function toScreen(v){ const p = v.clone().project(camera); return {x:(p.x+1)/2*innerWidth, y:(1-p.y)/2*innerHeight}; }
function turnBase(){            // tecla B: gira la capa inferior sola (prueba/ayuda)
  if (grab || snap) return;
  const {layer, pivot} = makePivot(1, -1);
  snap = {pivot, layer, axis:AX[1], from:0, to:-HALF, t0:performance.now(), dur:650};
}
const LOG = [];
const lg = (t, o) => { LOG.push(Object.assign({t:Math.round(performance.now()), e:t}, o||{})); if (LOG.length > 60) LOG.shift(); };
function grabStart(){
  if (snap){ lg("grab-ignorado-snap"); return; }
  const h = pick(cur.x, cur.y);
  if (h){ grab = {mode:"layer", h, x0:cur.x, y0:cur.y, axis:-1, x:cur.x}; setGlow([h.cubie]); lg("grab-capa", {x:Math.round(cur.x), y:Math.round(cur.y)}); }
  else { grab = {mode:"orbit", x:cur.x, y:cur.y}; lg("grab-fondo", {x:Math.round(cur.x), y:Math.round(cur.y)}); }
}
// elige eje de giro según hacia dónde arrastras: el punto tocado debe "seguir" al dedo
function lockAxis(dx, dy){
  const P = grab.h.point, P0 = toScreen(P);
  let best = null;
  for (let i = 0; i < 3; i++){
    if (i === grab.h.normalAxis) continue;
    const wa = AX[i].clone().applyQuaternion(cubeGroup.quaternion);
    const P1 = toScreen(P.clone().add(wa.clone().cross(P).multiplyScalar(0.05)));
    const sx = (P1.x-P0.x)/0.05, sy = (P1.y-P0.y)/0.05;       // píxeles por radian
    const len = Math.hypot(sx, sy) || 1e-6;
    const score = Math.abs(dx*sx + dy*sy)/len;
    if (!best || score > best.score) best = {i, sx, sy, len, score};
  }
  grab.axis = best.i; grab.sx = best.sx; grab.sy = best.sy; grab.len2 = best.len*best.len;
  const side = Math.sign(coord(grab.h.cubie, best.i)) || 1;
  const {layer, pivot} = makePivot(best.i, side);
  grab.layer = layer; grab.pivot = pivot; grab.angle = 0; lg("eje", {eje:best.i});
  setGlow(layer);
}
function grabEnd(){
  if (!grab) return;
  lg("suelta", {modo:grab.mode, eje:grab.axis, grados:Math.round((grab.angle||0)*57.3)});
  if (grab.mode === "layer" && grab.pivot){
    const a = grab.angle, n = Math.max(0, Math.ceil(Math.abs(a)/HALF - 0.06));  // en cuanto empiezas a girar, completa hasta 90°; arrastrando más, 180°…
    snap = {pivot:grab.pivot, layer:grab.layer, axis:AX[grab.axis], from:a, to:Math.sign(a)*n*HALF,
            t0:performance.now(), dur:260};
  }
  grab = null;
}
function stepSnap(now){
  if (!snap) return;
  const p = Math.min(1, (now - snap.t0)/snap.dur);
  snap.pivot.quaternion.setFromAxisAngle(snap.axis, snap.from + (snap.to - snap.from)*ease(p));
  if (p >= 1){
    snap.layer.forEach(c => {
      cubeGroup.attach(c);
      c.position.set(Math.round(c.position.x*2)/2, Math.round(c.position.y*2)/2, Math.round(c.position.z*2)/2);
    });
    cubeGroup.remove(snap.pivot); snap = null; setGlow(null);
  }
}

addEventListener("mousemove", e => {
  lastMouse = performance.now();
  if (lastMouse - lastHand > 1000){ target.x = e.clientX; target.y = e.clientY; if(!ptrVisible){cur.x=target.x;cur.y=target.y;} ptrVisible = true; }
});
addEventListener("mousedown", e => { if (performance.now() - lastHand > 1000){ target.x = cur.x = e.clientX; target.y = cur.y = e.clientY; pinched = true; } });
addEventListener("mouseup", () => { if (performance.now() - lastHand > 1000) pinched = false; });
addEventListener("keydown", e => { if (e.key==="b"||e.key==="B") turnBase(); });

function updatePointer(now){
  if (now - lastHand > (grab ? 1000 : 500) && now - lastMouse > 1000 && handSeenOnce){ ptrVisible = false; pinched = false; ratioS = -1; }
  cur.x += (target.x - cur.x) * (pinched ? 0.5 : 0.35); cur.y += (target.y - cur.y) * (pinched ? 0.5 : 0.35);
  ptrEl.style.display = ptrVisible ? "block" : "none";
  ptrEl.style.transform = `translate(${cur.x}px,${cur.y}px)`;
  ptrEl.classList.toggle("pinch", pinched);

  if (pinched && !wasPinched) grabStart();
  if (!pinched && wasPinched) grabEnd();
  wasPinched = pinched;

  if (grab){
    if (grab.mode === "layer"){
      const tx = cur.x - grab.x0, ty = cur.y - grab.y0;
      if (grab.axis < 0 && Math.hypot(tx, ty) > 14) lockAxis(tx, ty);
      if (grab.axis >= 0){
        grab.angle = (tx*grab.sx + ty*grab.sy)/grab.len2;
        grab.pivot.quaternion.setFromAxisAngle(AX[grab.axis], grab.angle);
      }
    } else {
      const dx = cur.x - grab.x, dy = cur.y - grab.y; grab.x = cur.x; grab.y = cur.y;
      yaw += dx*0.008; pitch = Math.max(-1.2, Math.min(1.2, pitch + dy*0.006));
    }
  } else if (ptrVisible && !snap){
    const h = pick(cur.x, cur.y); setGlow(h ? [h.cubie] : null);
  } else if (!ptrVisible && !snap) setGlow(null);
}

/* ---------- CÁMARA + MEDIAPIPE ---------- */
const dot = $("dot"), stxt = $("stxt");
function setStatus(t, cls){ stxt.textContent = t; dot.className = cls||""; }
let landmarker = null, lastVideoTime = -1, pinchFrames = 0, ratioS = -1;
const VER = "0.10.14";
const MODEL = "https://storage.googleapis.com/mediapipe-models/hand_landmarker/hand_landmarker/float16/1/hand_landmarker.task";

async function startCamera(){
  $("start").style.display = "none";
  try{
    const stream = await navigator.mediaDevices.getUserMedia({video:{width:640,height:480,facingMode:"user"},audio:false});
    video.srcObject = stream; video.style.display = "block"; await video.play();
  }catch(err){
    setStatus("Cámara no disponible: usa el ratón (o permite la cámara y recarga)", "err"); return;
  }
  setStatus("Cargando seguimiento de manos…");
  try{
    const {HandLandmarker, FilesetResolver} = await import(`https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@${VER}/vision_bundle.mjs`);
    const fs = await FilesetResolver.forVisionTasks(`https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@${VER}/wasm`);
    const opts = d => ({baseOptions:{modelAssetPath:MODEL, delegate:d}, runningMode:"VIDEO", numHands:1});
    try{ landmarker = await HandLandmarker.createFromOptions(fs, opts("GPU")); }
    catch(e){ landmarker = await HandLandmarker.createFromOptions(fs, opts("CPU")); }
    setStatus("Muestra tu mano a la cámara", "ok");
  }catch(err){
    console.error(err);
    setStatus("No se pudo cargar el seguimiento de manos (¿sin internet?). Usa el ratón", "err");
  }
}
$("go").addEventListener("click", startCamera);
$("mouse").addEventListener("click", () => { $("start").style.display="none"; setStatus("Modo ratón (prueba)"); });

const clamp = v => Math.max(0, Math.min(1, v));
function detectHand(now){
  if (!landmarker || video.readyState < 2 || video.currentTime === lastVideoTime) return;
  lastVideoTime = video.currentTime;
  const res = landmarker.detectForVideo(video, now);
  if (res.landmarks && res.landmarks.length){
    const L = res.landmarks[0], t4 = L[4], t8 = L[8];
    const palm = Math.hypot(L[5].x-L[17].x, L[5].y-L[17].y) || 0.1;
    const raw = Math.hypot(t4.x-t8.x, t4.y-t8.y) / palm;     // pulgar–índice relativo al tamaño de la mano
    ratioS = ratioS < 0 ? raw : ratioS*0.5 + raw*0.5; const ratio = ratioS;
    if (!pinched && ratio < 0.45){ if (++pinchFrames >= 2) { pinched = true; pinchFrames = 0; } }
    else if (pinched && ratio > 0.72){ if (++pinchFrames >= 3) { pinched = false; pinchFrames = 0; } }
    else pinchFrames = 0;
    // puntero = punto medio entre pulgar e índice (más estable que la punta sola)
    const mx = (t4.x + t8.x)/2, my = (t4.y + t8.y)/2;
    target.x = clamp(((1 - mx) - 0.10) / 0.80) * innerWidth;
    target.y = clamp((my - 0.08) / 0.78) * innerHeight;
    if (!ptrVisible){ cur.x = target.x; cur.y = target.y; }
    ptrVisible = true; lastHand = now; handSeenOnce = true;
    const ejes = ["X","Y","Z"];
    const est = (grab && grab.axis >= 0) ? `Girando capa sobre el eje ${ejes[grab.axis]}` : pinched ? "Agarrando" : "Mano detectada";
    setStatus(`${est} · pellizco ${ratio.toFixed(2)}`, "ok");
  } else if (now - lastHand > 500){
    setStatus("Muestra tu mano a la cámara", "ok");
  }
}

function resize(){
  const w = innerWidth, h = innerHeight;
  renderer.setSize(w, h, false);
  camera.aspect = w / h;
  if (camera.aspect < 1){ // vertical: más lejos para que quepa el cubo, y subido para dejar sitio al botón
    camera.position.z = Math.max(8, 6.8 / camera.aspect);
    camera.position.y = -camera.position.z * 0.09;
  } else { camera.position.z = 8; camera.position.y = 0; }
  camera.updateProjectionMatrix();
}
addEventListener("resize", resize); resize();

let prev = performance.now();
function frame(now){
  const dt = Math.min(100, now - prev); prev = now;
  detectHand(now);
  updatePointer(now);
  stepSnap(now);
  cubeGroup.rotation.set(pitch, yaw, 0);
  renderer.render(scene, camera);
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
