<script type="module">
/* ================== PREGUNTAS ================== */
const PREGUNTAS = [
  {
    texto: "¿Cuál es el planeta más grande del sistema solar?",
    opciones: ["Saturno", "Júpiter", "La Tierra", "Neptuno"],
    correcta: 1,
    explicacion: "Júpiter es el mayor planeta: en su interior cabrían más de 1.300 Tierras. Saturno es el segundo."
  },
  {
    texto: "¿A qué temperatura hierve el agua al nivel del mar?",
    opciones: ["0 °C", "50 °C", "80 °C", "100 °C"],
    correcta: 3,
    explicacion: "A la presión del nivel del mar (1 atmósfera) el agua hierve a 100 °C. A 0 °C se congela. En la montaña, con menos presión, hierve antes."
  },
  {
    texto: "¿Qué gas toman las plantas del aire para hacer la fotosíntesis?",
    opciones: ["Dióxido de carbono", "Oxígeno", "Nitrógeno", "Helio"],
    correcta: 0,
    explicacion: "Las plantas absorben dióxido de carbono (CO₂) y, con la luz del sol y el agua, fabrican azúcares y liberan oxígeno."
  }
];
const LETRAS = ["A","B","C","D"];
const DWELL_MS = 500;

/* ================== ESTADO DEL EXAMEN ================== */
let indice = 0, seleccion = null, fase = "elegir", aciertos = 0, historial = [];
const $ = id => document.getElementById(id);

function mostrar(id){
  document.querySelectorAll(".pantalla").forEach(p => p.classList.remove("activa"));
  $(id).classList.add("activa");
}

function pintarPregunta(){
  const q = PREGUNTAS[indice];
  seleccion = null; fase = "elegir";
  $("progreso").textContent = `Pregunta ${indice+1} de ${PREGUNTAS.length}`;
  $("enunciado").textContent = q.texto;
  const cont = $("opciones"); cont.innerHTML = "";
  q.opciones.forEach((txt,i) => {
    const b = document.createElement("button");
    b.className = "opcion"; b.dataset.dwell = ""; b.dataset.i = i;
    b.innerHTML = `<span class="letra">${LETRAS[i]}</span><span>${txt}</span>`;
    b.addEventListener("click", () => elegir(i));
    cont.appendChild(b);
  });
  const f = $("feedback"); f.className = "feedback"; f.innerHTML = "";
  const acc = $("btn-accion"); acc.textContent = "CONTESTAR"; acc.disabled = true;
  mostrar("p-pregunta");
}

function elegir(i){
  if (fase !== "elegir") return;
  seleccion = i;
  document.querySelectorAll(".opcion").forEach(o =>
    o.classList.toggle("seleccionada", Number(o.dataset.i) === i));
  $("btn-accion").disabled = false;
}

function contestar(){
  if (fase !== "elegir" || seleccion === null) return;
  const q = PREGUNTAS[indice];
  const ok = seleccion === q.correcta;
  if (ok) aciertos++;
  historial.push({ q: q.texto, ok, elegida: q.opciones[seleccion], buena: q.opciones[q.correcta] });
  fase = "corregida";
  document.querySelectorAll(".opcion").forEach(o => {
    const i = Number(o.dataset.i);
    o.classList.add("bloqueada");
    o.classList.remove("seleccionada");
    if (i === q.correcta) o.classList.add("correcta");
    else if (i === seleccion) o.classList.add("incorrecta");
  });
  const f = $("feedback");
  f.className = "feedback " + (ok ? "ok" : "ko");
  f.innerHTML = ok
    ? `<strong>✅ ¡Correcto!</strong>${q.explicacion}`
    : `<strong>❌ No es correcta.</strong>La respuesta buena es <b>${q.opciones[q.correcta]}</b>. ${q.explicacion}`;
  $("btn-accion").textContent = indice < PREGUNTAS.length-1 ? "SIGUIENTE PREGUNTA" : "VER PUNTUACIÓN";
}

function accion(){
  if (fase === "elegir") contestar();
  else { indice++; indice < PREGUNTAS.length ? pintarPregunta() : pintarFinal(); }
}

function pintarFinal(){
  $("nota").textContent = `${aciertos} / ${PREGUNTAS.length}`;
  $("mensaje-final").textContent =
    aciertos === 3 ? "¡Perfecto! Dominas la ciencia básica." :
    aciertos === 2 ? "¡Muy bien! Solo se te escapó una." :
    aciertos === 1 ? "Vas por buen camino. Repasa las explicaciones." :
                     "Ánimo: lee las explicaciones y vuelve a intentarlo.";
  $("resumen").innerHTML = historial.map((h,i) =>
    `<div class="${h.ok?'r-ok':'r-ko'}"><b>${i+1}.</b> ${h.q}<br>${h.ok?'✅':'❌'} Tu respuesta: ${h.elegida}${h.ok?'':` · Correcta: <b>${h.buena}</b>`}</div>`).join("");
  mostrar("p-final");
}

function reiniciar(){ indice = 0; aciertos = 0; historial = []; pintarPregunta(); }

$("btn-accion").addEventListener("click", accion);
$("btn-repetir").addEventListener("click", reiniciar);
$("btn-camara").addEventListener("click", iniciarCamara);

/* ================== CÁMARA + MEDIAPIPE ================== */
const video = $("video"), esq = $("esqueleto"), ectx = esq.getContext("2d");
const capa = $("capa"), ctx = capa.getContext("2d");
let detector = null, camaraLista = false;

function ajustarCapa(){
  capa.width = innerWidth * devicePixelRatio; capa.height = innerHeight * devicePixelRatio;
  capa.style.width = innerWidth+"px"; capa.style.height = innerHeight+"px";
  ctx.setTransform(devicePixelRatio,0,0,devicePixelRatio,0,0);
}
addEventListener("resize", ajustarCapa); ajustarCapa();

async function iniciarCamara(){
  if (camaraLista) { pintarPregunta(); return; }
  const est = $("estado");
  try {
    est.textContent = "Cargando el detector de manos…";
    const vision = await import("https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@0.10.14/vision_bundle.mjs");
    const fileset = await vision.FilesetResolver.forVisionTasks(
      "https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@0.10.14/wasm");
    const opciones = (delegate) => ({
      baseOptions: {
        modelAssetPath: "https://storage.googleapis.com/mediapipe-models/hand_landmarker/hand_landmarker/float16/1/hand_landmarker.task",
        delegate
      },
      runningMode: "VIDEO", numHands: 2,
      minHandDetectionConfidence: 0.6, minHandPresenceConfidence: 0.6, minTrackingConfidence: 0.5
    });
    try { detector = await vision.HandLandmarker.createFromOptions(fileset, opciones("GPU")); }
    catch { detector = await vision.HandLandmarker.createFromOptions(fileset, opciones("CPU")); }

    est.textContent = "Pidiendo permiso para la cámara…";
    const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "user", width: 640, height: 480 }, audio: false });
    video.srcObject = stream; await video.play();
    $("mini").style.display = "block";
    camaraLista = true;
    requestAnimationFrame(bucle);
    pintarPregunta();
  } catch (e) {
    console.error(e);
    est.textContent = "No se pudo activar la cámara o el detector. Abre la página por HTTPS y concede permiso. (Puedes jugar con el ratón).";
    setTimeout(pintarPregunta, 2500);
  }
}

/* ---------- elegir la mano según el selector ----------
   MediaPipe etiqueta la lateralidad como si la imagen estuviera en espejo:
   con la cámara frontal sin voltear, tu mano DERECHA llega etiquetada "Left". */
function elegirMano(res){
  if (!res.landmarks || !res.landmarks.length) return null;
  const modo = $("sel-mano").value;
  if (modo === "cualquiera") return res.landmarks[0];
  const buscada = modo === "derecha" ? "Left" : "Right";
  for (let i = 0; i < res.landmarks.length; i++){
    const lab = res.handednesses?.[i]?.[0]?.categoryName || res.handedness?.[i]?.[0]?.categoryName;
    if (lab === buscada) return res.landmarks[i];
  }
  return null;
}

/* ---------- puntero suavizado y permanencia ---------- */
let px = null, py = null;
let objetivo = null, tInicio = 0, bloqueado = null; // bloqueado: elemento ya activado, hay que salir de él
const SUAVE = 0.35, MARGEN = 0.12; // margen: no hace falta llevar el dedo al borde de la cámara

function mapear(lm){
  // espejo + ampliación del área útil
  let nx = 1 - lm.x, ny = lm.y;
  nx = (nx - MARGEN) / (1 - 2*MARGEN);
  ny = (ny - MARGEN) / (1 - 2*MARGEN);
  nx = Math.min(1, Math.max(0, nx)); ny = Math.min(1, Math.max(0, ny));
  return [nx * innerWidth, ny * innerHeight];
}

function elementoActivable(x, y){
  const el = document.elementFromPoint(x, y);
  const t = el && el.closest("[data-dwell]");
  if (!t || t.disabled || t.classList.contains("bloqueada")) return null;
  if (!t.closest(".pantalla.activa")) return null;
  return t;
}

function marcarHover(el){
  document.querySelectorAll(".hover").forEach(h => { if (h !== el) h.classList.remove("hover"); });
  if (el) el.classList.add("hover");
}

let ultimoTs = -1;
function bucle(){
  requestAnimationFrame(bucle);
  if (!detector || video.readyState < 2) return;
  const ts = performance.now();
  if (ts === ultimoTs) return; ultimoTs = ts;
  const res = detector.detectForVideo(video, ts);
  dibujarEsqueleto(res);

  ctx.clearRect(0, 0, innerWidth, innerHeight);
  const mano = elegirMano(res);
  if (!mano){ px = py = null; objetivo = null; marcarHover(null); return; }

  const [tx, ty] = mapear(mano[8]); // punta del índice
  if (px === null){ px = tx; py = ty; }
  px += (tx - px) * SUAVE; py += (ty - py) * SUAVE;

  const el = elementoActivable(px, py);
  if (bloqueado && el !== bloqueado) bloqueado = null;
  const candidato = (el && el !== bloqueado) ? el : null;

  if (candidato !== objetivo){ objetivo = candidato; tInicio = ts; }
  marcarHover(candidato);

  let progreso = 0;
  if (objetivo){
    progreso = Math.min(1, (ts - tInicio) / DWELL_MS);
    if (progreso >= 1){
      const activado = objetivo;
      bloqueado = activado; objetivo = null;
      activado.click();
      destello(px, py);
    }
  }
  dibujarPuntero(px, py, progreso);
}

/* ---------- dibujo del puntero con círculo creciente ---------- */
let flash = null;
function destello(x, y){ flash = { x, y, t: performance.now() }; }

function dibujarPuntero(x, y, p){
  // círculo creciente (relleno) + arco de progreso
  if (p > 0){
    const r = 10 + 26 * p;
    ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI*2);
    ctx.fillStyle = `rgba(47,111,237,${0.15 + 0.25*p})`; ctx.fill();
    ctx.beginPath(); ctx.arc(x, y, 38, -Math.PI/2, -Math.PI/2 + Math.PI*2*p);
    ctx.strokeStyle = "#2f6fed"; ctx.lineWidth = 5; ctx.lineCap = "round"; ctx.stroke();
  }
  // punto del índice
  ctx.beginPath(); ctx.arc(x, y, 8, 0, Math.PI*2);
  ctx.fillStyle = "#ff7a00"; ctx.fill();
  ctx.lineWidth = 3; ctx.strokeStyle = "#fff"; ctx.stroke();
  // destello al confirmar
  if (flash){
    const k = (performance.now() - flash.t) / 350;
    if (k < 1){
      ctx.beginPath(); ctx.arc(flash.x, flash.y, 38 + 30*k, 0, Math.PI*2);
      ctx.strokeStyle = `rgba(31,157,85,${1-k})`; ctx.lineWidth = 6; ctx.stroke();
    } else flash = null;
  }
}

/* ---------- esqueleto en la miniatura ---------- */
const UNIONES = [[0,1],[1,2],[2,3],[3,4],[0,5],[5,6],[6,7],[7,8],[5,9],[9,10],[10,11],[11,12],
                 [9,13],[13,14],[14,15],[15,16],[13,17],[17,18],[18,19],[19,20],[0,17]];
function dibujarEsqueleto(res){
  esq.width = esq.clientWidth; esq.height = esq.clientHeight;
  ectx.clearRect(0, 0, esq.width, esq.height);
  const activa = elegirMano(res);
  (res.landmarks || []).forEach(lm => {
    const esActiva = lm === activa;
    ectx.strokeStyle = esActiva ? "#00e5ff" : "rgba(255,255,255,.45)";
    ectx.lineWidth = esActiva ? 3 : 2;
    UNIONES.forEach(([a,b]) => {
      ectx.beginPath();
      ectx.moveTo(lm[a].x*esq.width, lm[a].y*esq.height);
      ectx.lineTo(lm[b].x*esq.width, lm[b].y*esq.height);
      ectx.stroke();
    });
    if (esActiva){
      ectx.beginPath(); ectx.arc(lm[8].x*esq.width, lm[8].y*esq.height, 6, 0, Math.PI*2);
      ectx.fillStyle = "#ff7a00"; ectx.fill();
    }
  });
}
