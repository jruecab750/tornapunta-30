/* ================= SNAP (imán) ================= */
function getSnaps(){
  if(snapCache && snapCache.v===geoVer+"|"+marks.length+"|"+lines.length) return snapCache.list;
  const list=[];
  poly.forEach(p=>list.push({x:p.x,y:p.y}));
  marks.forEach(m=>list.push({x:m.x,y:m.y}));
  for(const l of lines){
    for(let i=0;i<poly.length;i++){ const h=lineSeg(l,poly[i],poly[(i+1)%poly.length]); if(h) list.push({x:h.x,y:h.y}); }
  }
  for(let i=0;i<lines.length;i++) for(let j=i+1;j<lines.length;j++){
    const p=lineLine(lines[i],lines[j]); if(p&&inPoly(p,poly,1e-4)) list.push(p);
  }
  snapCache={v:geoVer+"|"+marks.length+"|"+lines.length,list};
  return list;
}
function nearestSnap(){
  let best=null, bd=1e9;
  for(const t of getSnaps()){
    const s=boardToScreen(t.x,t.y), d=Math.hypot(s.x-cur.x,s.y-cur.y);
    const lim=(lastSnap&&Math.hypot(lastSnap.x-t.x,lastSnap.y-t.y)<1e-6)?26:17;
    if(d<lim&&d<bd){bd=d;best=t;}
  }
  lastSnap=best; return best;
}

function setTool(t){
  if(stage!=="trace") return;
  tool=t; anchor=null; tapeStart=null; sawLock=null; sawTilt=null; tapeDraw=null; sideDirty=true;
}
function undo(){
  if(stage!=="trace") return;
  const h=history.pop(); if(!h){ say("No hay nada que deshacer"); return; }
  if(h.t==="mark") marks=marks.filter(m=>m.id!==h.id); else lines=lines.filter(l=>l.id!==h.id);
  anchor=null; tapeStart=null; geoVer++; say("Deshecho","");
}
function addMark(x,y){
  const e=marks.find(m=>Math.hypot(m.x-x,m.y-y)<0.05); if(e) return e;
  const m={id:idc++,x:Math.round(x*1000)/1000,y:Math.round(y*1000)/1000}; marks.push(m); history.push({t:"mark",id:m.id}); geoVer++; return m;
}
function addLine(x,y,ang){
  const dup=lines.find(l=>angDiff(l.ang,ang)<0.01&&distPtLine({x,y},l)<0.02);
  if(dup){ say("Esa línea ya está trazada"); return dup; }
  const l={id:idc++,x,y,ang}; lines.push(l); history.push({t:"line",id:l.id}); geoVer++; return l;
}
const matchCut=(l,c)=>angDiff(l.ang,c.ang)<0.6 && distPtLine({x:c.x,y:c.y},l)<TOL;

function doCut(){
  const l=lines.find(x=>x.id===sawLock); if(!l) return;
  if(sawTilt==null){ say("Elige primero la inclinación de la hoja","err"); return; }
  const c=CUTS.find(c=>matchCut(l,c));
  if(!c){ errors++; say("Ese trazado no es un corte del tornapunta. Revisa el método (pasos 4 a 7).","err",6000); return; }
  if(cutsDone.includes(c.n)){ say("Ese corte ya está hecho","err"); return; }
  const next=[4,5,6,7].find(n=>!cutsDone.includes(n));
  if(c.n!==next){ errors++; say(`Orden de los cortes: 4, 5, 6 y 7. Ahora toca el corte ${next}.`,"err",6000); return; }
  if(sawTilt!==90){ errors++; say(`Con la hoja a ${sawTilt}° el corte sale a bisel y el apoyo no queda plano. Hazlo a 90° (vertical).`,"err",7000); return; }
  const A=clipHalf(poly,l,1), B=clipHalf(poly,l,-1);
  const keep=area(A)>=area(B)?A:B, drop=keep===A?B:A;
  dropPiece(drop); poly=keep; rebuildBoard();
  cutsDone.push(c.n); sawLock=null; sawTilt=null; geoVer++; sideDirty=true;
  say(`Corte ${c.n} correcto ✓`,"ok");
}

let pb=null;                          // puntero en coordenadas de tabla (cm)
function computeHover(){
  hv={}; tapeDraw=null; snapShow=null;
  if($("start").style.display!=="none") return null;
  for(const el of document.querySelectorAll(".dw")){
    if(!el.getClientRects().length) continue;
    const r=el.getBoundingClientRect();
    if(cur.x>=r.left-6&&cur.x<=r.right+6&&cur.y>=r.top-6&&cur.y<=r.bottom+6) return {key:el.dataset.key,dom:true,el};
  }
  if(stage!=="trace"||!ptrVisible||!pb) return null;

  if(tool==="pencil"){
    if(anchor){
      const v={x:pb.x-anchor.x,y:pb.y-anchor.y}, dist=Math.hypot(v.x,v.y);
      if(dist<1.2) return {key:"anchorcancel"};
      let a=Math.atan2(v.y,v.x)*180/Math.PI; if(a<0)a+=180; if(a>=180)a-=180;
      let best=null,bd=99; for(const c of CAND){const d=angDiff(a,c); if(d<bd){bd=d;best=c;}}
      if(bd<7){ hv.rayAng=best; return {key:"ray:"+best,ang:best}; }
    }
    const s=nearestSnap(); snapShow=s;
    if(s) return {key:"pt:"+s.x.toFixed(2)+","+s.y.toFixed(2),pt:s};
    if(inPoly(pb,poly,0.05)) return {key:"pt:free",pt:{x:Math.round(pb.x*20)/20,y:Math.round(pb.y*20)/20}};
    return null;
  }
  if(tool==="flex"){
    if(!tapeStart){
      const s=nearestSnap(); snapShow=s;
      if(s) return {key:"fs:"+s.x.toFixed(2)+","+s.y.toFixed(2),pt:s};
      if(inPoly(pb,poly,0.05)) return {key:"fs:free",pt:{x:Math.round(pb.x*20)/20,y:Math.round(pb.y*20)/20}};
      return null;
    }
    const S=tapeStart, v={x:pb.x-S.x,y:pb.y-S.y};
    const dirs=[0,90]; lines.forEach(l=>{ if(distPtLine(S,l)<0.05 && !dirs.some(d=>angDiff(d,l.ang)<0.01)) dirs.push(l.ang); });
    let a=Math.atan2(v.y,v.x)*180/Math.PI; if(a<0)a+=180; if(a>=180)a-=180;
    let best=null,bd=99; for(const d of dirs){const dd=angDiff(a,d); if(dd<bd){bd=dd;best=d;}}
    if(Math.hypot(v.x,v.y)<0.8||bd>8) { tapeDraw={s:S,a:null}; return {key:"fcancel"}; }
    const d=dirOf(best), t=v.x*d.x+v.y*d.y, ts=Math.round(t/0.5)*0.5;
    tapeDraw={s:S,a:best,t:ts};
    if(Math.abs(ts)<0.5) return {key:"fcancel"};
    const e={x:S.x+d.x*ts,y:S.y+d.y*ts};
    return {key:"fe:"+best+":"+ts,pt:e};
  }
  if(tool==="saw"){
    if(sawLock) return null;
    let best=null,bd=0.9;
    for(const l of lines){
      const seg=clipLine(l,poly); if(!seg) continue;
      const d=distPtLine(pb,l); if(d>=bd) continue;
      const dd=dirOf(l.ang), t=(pb.x-l.x)*dd.x+(pb.y-l.y)*dd.y;
      if(t<seg[0].t-1||t>seg[1].t+1) continue;
      bd=d; best=l;
    }
    if(best){ hv.lineId=best.id; return {key:"sl:"+best.id,lineId:best.id}; }
  }
  return null;
}

function fire(h){
  const k=h.key;
  if(k.startsWith("tool:")) setTool(k.slice(5));
  else if(k==="undo") undo();
  else if(k==="front"){ for(const a of AXN) U[a]-=wrapA(U[a]); say("Vista frontal: ejes X, Y, Z a 0°","ok",2000); }
  else if(k==="reset") resetAll();
  else if(k.startsWith("tilt:")){ sawTilt=+k.slice(5); sideDirty=true; }
  else if(k==="cut") doCut();
  else if(k==="sawcancel"){ sawLock=null; sawTilt=null; sideDirty=true; }
  else if(k==="mount") startMount();
  else if(k.startsWith("pt:")){ const pt=(k==="pt:free")?{x:h.pt.x,y:h.pt.y}:h.pt; const m=addMark(pt.x,pt.y); anchor={x:m.x,y:m.y}; say("Punto marcado. Ahora elige una línea: apunta en su dirección y pellizca.","ok",3500); }
  else if(k==="anchorcancel"){ anchor=null; }
  else if(k.startsWith("ray:")){ addLine(anchor.x,anchor.y,h.ang); anchor=null; say("Línea trazada","ok",2000); }
  else if(k.startsWith("fs:")){ tapeStart={x:h.pt.x,y:h.pt.y}; say("Alinea el flexómetro con una línea y avanza; pellizca para marcar el final de la medida.","",5000); }
  else if(k==="fcancel"){ tapeStart=null; }
  else if(k.startsWith("fe:")){ const m=addMark(h.pt.x,h.pt.y); tapeStart=null; say(`Medida marcada: ${f1(Math.abs(parseFloat(k.split(":")[2])))} cm`,"ok",3000); }
  else if(k.startsWith("sl:")){ sawLock=h.lineId; sideDirty=true; say("Hoja alineada. Elige la inclinación de la hoja y pulsa CORTAR.","",6000); }
}

const DRAG_PX=28, ARM_MS=400, GAIN_HAND=0.6, RESET_MS=1500;
function refreshHover(){ pb=(ptrVisible&&stage==="trace")?screenToBoard(cur.x,cur.y):null; H0=computeHover(); hv.dom=H0&&H0.dom; }
const padEl=document.getElementById("orbit");
function padGeo(){ const r=padEl.getBoundingClientRect(); return {cx:r.left+r.width/2,cy:r.top+r.height/2,r:r.width/2}; }
function padAt(x,y){ const g=padGeo(); return Math.hypot(x-g.cx,y-g.cy)<=g.r; }
function pressStart(now){
  const hand=now-lastHand<1000;
  let pos={x:cur.x,y:cur.y};
  if(hand){ for(let i=hist.length-1;i>=0;i--){ pos=hist[i]; if(now-hist[i].t>=150) break; } }   // posición previa al gesto de pellizcar
  const raw={x:target.x,y:target.y};
  cur.x=pos.x; cur.y=pos.y; refreshHover();
  press={t:now,raw,pos:{x:pos.x,y:pos.y},hand,mode:"pending",h:H0};
  const sl=sliderAt(cur.x,cur.y);
  if(sl){ press.mode="sliderArm"; press.sl=sl; press.at=hand?0:ARM_MS; }
  else if(stage==="trace"&&padAt(cur.x,cur.y)){
    const g=padGeo(); press.mode="orbit"; press.base={x:U.x,y:U.y,z:U.z}; press.n={x:0,y:0,z:0};
    press.ring=Math.hypot(cur.x-g.cx,cur.y-g.cy)>g.r*0.68; padEl.classList.add("on");
  }
}
function pressEnd(now){
  const P=press; press=null; if(!P) return;
  padEl.classList.remove("on");
  if(P.sl){ P.sl.el.classList.remove("on"); P.sl.el.querySelector(".thumb").style.transform=""; }
  if(P.mode==="pending"){
    const h=P.h;
    if(h&&h.key){
      if(h.key==="reset"&&now-P.t<RESET_MS) say("Para reiniciar, mantén el pellizco 1,5 s","err",2500);
      else fire(h);
    }
  }
}
function dragStart(){
  const P=press; P.mode="drag"; P.axis=-1; P.n=0; P.base={x:U.x,y:U.y,z:U.z};
  P.cen=boardToScreen(L/2,HH/2); const e=boardToScreen(L,HH/2); P.R=Math.max(60,Math.hypot(e.x-P.cen.x,e.y-P.cen.y));
}
function inputStep(now,dt){
  ringP=0;
  if(pinched&&!wasPinched) pressStart(now);
  if(!pinched&&wasPinched) pressEnd(now);
  wasPinched=pinched;
  ptrEl.classList.toggle("pinch",pinched);
  const P=press;
  if(!P){
    const sl=(ptrVisible&&!(H0&&H0.dom))?sliderAt(cur.x,cur.y):null;
    SLIDERS.forEach(o=>o.el.classList.toggle("hover",o===sl));
    padEl.classList.toggle("hover",ptrVisible&&stage==="trace"&&padAt(cur.x,cur.y));
    return;
  }
  if(P.mode==="sliderArm"){
    if(sliderAt(cur.x,cur.y)!==P.sl){ P.mode="none"; P.sl.el.classList.remove("on"); P.sl.el.querySelector(".thumb").style.transform=""; }
    else{
      P.at+=dt; const p=Math.min(1,P.at/ARM_MS); ringP=p;
      P.sl.el.querySelector(".thumb").style.transform=`scale(${1+0.45*p})`;
      if(p>=1){ P.mode="slider"; P.x0=cur.x; P.v0=P.sl.get(); P.sl.el.classList.add("on"); }
    }
  } else if(P.mode==="slider"){
    const r=P.sl.el.getBoundingClientRect(), gain=P.hand?GAIN_HAND:1;
    setSlider(P.sl,P.v0+(cur.x-P.x0)/r.width*(P.sl.max-P.sl.min)*gain);
  } else if(P.mode==="pending"){
    P.h=H0;
    if(P.h&&P.h.key==="reset") ringP=Math.min(1,(now-P.t)/RESET_MS);
  } else if(P.mode==="orbit"){
    const g=padGeo();
    if(Math.hypot(cur.x-g.cx,cur.y-g.cy)>g.r+8){                 // al salir de la esfera, la tabla deja de moverse
      P.mode="none"; padEl.classList.remove("on"); say("Fuera de la esfera: la tabla queda quieta","",1600);
    } else if(stage==="trace"){
      const stepPx=700/sensPct, dx=target.x-P.raw.x, dy=target.y-P.raw.y, out={};
      if(P.ring){                                                  // anillo exterior: giro en Z (como un volante)
        const rx=P.pos.x-g.cx, ry=-(P.pos.y-g.cy), rl=Math.hypot(rx,ry)||1;
        out.z=(rx*(-dy)-ry*dx)/rl;
      } else { out.x=dy; out.y=dx; }                               // disco: órbita (abajo = +X, derecha = +Y)
      for(const a of AXN){ if(out[a]===undefined) continue;
        const u=out[a]/stepPx; if(Math.abs(u-P.n[a])>0.62) P.n[a]=Math.round(u);
        const nv=P.base[a]+P.n[a]*15;
        if(nv!==U[a]){ U[a]=nv; say(`Eje ${a.toUpperCase()}: ${fmtA(wrapA(nv))}`,"",1400); }
      }
    }
  }
}
